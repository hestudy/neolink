#### OpenAI 集成规范

[Source: docs/architecture.md#AI 服务集成架构]

**技术栈**: OpenAI SDK + TypeScript
**主要模型**: GPT-4o-mini (成本效益最佳)
**用途**: 内容摘要、标签生成

**实现要求**:

```typescript
// packages/ai/src/providers/OpenAIProvider.ts
import { OpenAI } from 'openai';
import { AIProvider, SummaryOptions, SummaryResult } from '../types';

export class OpenAIProvider implements AIProvider {
  private client: OpenAI;
  private cache: CacheService;
  private costTracker: CostTracker;

  constructor(config: OpenAIConfig) {
    this.client = new OpenAI({
      apiKey: config.apiKey,
      timeout: 30000,
      maxRetries: 3,
    });
    this.cache = new CacheService(config.redis);
    this.costTracker = new CostTracker(config.costLimits);
  }

  async generateSummary(
    content: string,
    options: SummaryOptions = {}
  ): Promise<SummaryResult> {
    // 1. 成本检查
    await this.costTracker.checkBudget('summary');

    // 2. 缓存检查
    const cacheKey = this.generateCacheKey('summary', content, options);
    const cached = await this.cache.get(cacheKey);
    if (cached) return cached;

    // 3. API调用
    const response = await this.client.chat.completions.create({
      model: 'gpt-4o-mini',
      messages: [
        {
          role: 'system',
          content: this.getSummarySystemPrompt(options),
        },
        {
          role: 'user',
          content: this.preprocessContent(content, options.maxLength || 4000),
        },
      ],
      max_tokens:
        options.summaryLength === 'short'
          ? 100
          : options.summaryLength === 'long'
            ? 300
            : 200,
      temperature: 0.3,
    });

    // 4. 处理响应和缓存
    const summary = response.choices[0]?.message?.content?.trim();
    if (!summary) {
      throw new AIError('Failed to generate summary');
    }

    // 5. 成本记录
    await this.costTracker.recordUsage('summary', {
      inputTokens: response.usage?.prompt_tokens || 0,
      outputTokens: response.usage?.completion_tokens || 0,
      cost: this.calculateCost(response.usage),
    });

    const result: SummaryResult = {
      summary,
      confidence: this.calculateConfidence(response),
      language: await this.detectLanguage(summary),
    };

    await this.cache.set(cacheKey, result, { ttl: 7 * 24 * 3600 });
    return result;
  }

  async generateTags(
    content: string,
    options: TagOptions = {}
  ): Promise<TagResult> {
    // 类似的实现模式
  }
}
```

import { Anthropic } from '@anthropic-ai/sdk';
import { AIProvider, SummaryOptions, SummaryResult } from '../types';

export class ClaudeProvider implements AIProvider {
  private client: Anthropic;
  private cache: CacheService;
  private costTracker: CostTracker;

  constructor(config: ClaudeConfig) {
    this.client = new Anthropic({
      apiKey: config.apiKey,
      timeout: 30000,
    });
    this.cache = new CacheService(config.redis);
    this.costTracker = new CostTracker(config.costLimits);
  }

  async generateSummary(
    content: string,
    options: SummaryOptions = {}
  ): Promise<SummaryResult> {
    await this.costTracker.checkBudget('summary');

    const cacheKey = this.generateCacheKey('summary', content, options);
    const cached = await this.cache.get(cacheKey);
    if (cached) return cached;

    const response = await this.client.messages.create({
      model: 'claude-3-haiku-20240307',
      max_tokens:
        options.summaryLength === 'short'
          ? 100
          : options.summaryLength === 'long'
            ? 300
            : 200,
      messages: [
        {
          role: 'user',
          content: `${this.getSummarySystemPrompt(options)}\n\n${content}`,
        },
      ],
      temperature: 0.3,
    });

    const summary = response.content[0]?.text?.trim();
    if (!summary) {
      throw new AIError('Failed to generate summary with Claude');
    }

    // 成本记录和缓存逻辑
    await this.costTracker.recordUsage('summary', {
      inputTokens: response.usage?.input_tokens || 0,
      outputTokens: response.usage?.output_tokens || 0,
      cost: this.calculateClaudeCost(response.usage),
    });

    const result: SummaryResult = {
      summary,
      confidence: this.calculateConfidence(response),
      language: await this.detectLanguage(summary),
    };

    await this.cache.set(cacheKey, result, { ttl: 7 * 24 * 3600 });
    return result;
  }
}

#### AI 服务统一接口

[Source: docs/architecture.md#AI 服务集成架构]

**设计模式**: Strategy Pattern + Factory Pattern
**目的**: 提供统一的 AI 调用接口，支持多提供商和故障转移

```typescript
// packages/ai/src/services/AIService.ts
export class AIService {
  private providers: Map<string, AIProvider>;
  private config: AIConfig;
  private fallbackChain: string[];

  constructor(config: AIConfig) {
    this.config = config;
    this.providers = new Map();
    this.fallbackChain = ['openai', 'claude'];

    // 初始化提供商
    if (config.openai?.enabled) {
      this.providers.set('openai', new OpenAIProvider(config.openai));
    }
    if (config.claude?.enabled) {
      this.providers.set('claude', new ClaudeProvider(config.claude));
    }
  }

  async generateSummary(
    content: string,
    options: SummaryOptions = {}
  ): Promise<SummaryResult> {
    const provider =
      options.provider || this.config.defaultProvider || 'openai';
    return this.executeWithFallback(
      'generateSummary',
      provider,
      content,
      options
    );
  }

  private async executeWithFallback<T>(
    method: keyof AIProvider,
    primaryProvider: string,
    ...args: any[]
  ): Promise<T> {
    const providers = [
      primaryProvider,
      ...this.fallbackChain.filter((p) => p !== primaryProvider),
    ];

    for (const providerName of providers) {
      const provider = this.providers.get(providerName);
      if (!provider) continue;

      try {
        return await (provider[method] as any)(...args);
      } catch (error) {
        console.warn(`AI provider ${providerName} failed:`, error.message);

        // 如果是成本限制错误，不要fallback
        if (error instanceof BudgetExceededError) {
          throw error;
        }

        continue;
      }
    }

    throw new AIError('All AI providers failed');
  }
}
```

#### 成本跟踪和预算控制

[Source: docs/architecture.md#AI 服务集成架构]

**目标**: 每用户每月 AI 成本 < $2
**机制**: 实时成本跟踪、预算警告、自动限制

```typescript
// packages/ai/src/services/CostTracker.ts
interface CostLimits {
  monthlyBudget: number;
  dailyBudget: number;
  userBudget: number;
}

interface UsageRecord {
  inputTokens: number;
  outputTokens: number;
  cost: number;
  timestamp: Date;
  model: string;
  operation: string;
}

export class CostTracker {
  private redis: Redis;
  private limits: CostLimits;

  constructor(redis: Redis, limits: CostLimits) {
    this.redis = redis;
    this.limits = limits;
  }

  async checkBudget(operation: string, userId?: string): Promise<void> {
    const currentMonth = new Date().toISOString().slice(0, 7); // YYYY-MM
    const currentDay = new Date().toISOString().slice(0, 10); // YYYY-MM-DD

    // 检查月度预算
    const monthlySpend = await this.getSpend(`monthly:${currentMonth}`);
    if (monthlySpend >= this.limits.monthlyBudget) {
      throw new BudgetExceededError('Monthly budget exceeded');
    }

    // 检查日度预算
    const dailySpend = await this.getSpend(`daily:${currentDay}`);
    if (dailySpend >= this.limits.dailyBudget) {
      throw new BudgetExceededError('Daily budget exceeded');
    }

    // 检查用户预算（如果有用户ID）
    if (userId) {
      const userSpend = await this.getSpend(`user:${userId}:${currentMonth}`);
      if (userSpend >= this.limits.userBudget) {
        throw new BudgetExceededError('User budget exceeded');
      }
    }
  }

  async recordUsage(
    operation: string,
    usage: UsageRecord,
    userId?: string
  ): Promise<void> {
    const currentMonth = new Date().toISOString().slice(0, 7);
    const currentDay = new Date().toISOString().slice(0, 10);

    // 记录各级别的支出
    await this.addSpend(`monthly:${currentMonth}`, usage.cost);
    await this.addSpend(`daily:${currentDay}`, usage.cost);

    if (userId) {
      await this.addSpend(`user:${userId}:${currentMonth}`, usage.cost);
    }

    // 记录详细使用记录
    const record = {
      ...usage,
      operation,
      userId,
      timestamp: new Date(),
    };

    await this.redis.lpush(`usage_log:${currentMonth}`, JSON.stringify(record));
  }

  private async getSpend(key: string): Promise<number> {
    const spend = await this.redis.get(`spend:${key}`);
    return spend ? parseFloat(spend) : 0;
  }

  private async addSpend(key: string, amount: number): Promise<void> {
    await this.redis.incrbyfloat(`spend:${key}`, amount);
    await this.redis.expire(`spend:${key}`, 32 * 24 * 3600); // 32天过期
  }
}
```

#### 缓存策略

[Source: docs/architecture.md#AI 服务集成架构]

**目标**: 避免重复 AI 调用，降低成本
**策略**: 基于内容哈希的智能缓存，TTL 策略

```typescript
// packages/ai/src/utils/cache.ts
export class CacheService {
  private redis: Redis;

  constructor(redis: Redis) {
    this.redis = redis;
  }

  async get<T>(key: string): Promise<T | null> {
    const cached = await this.redis.get(key);
    return cached ? JSON.parse(cached) : null;
  }

  async set<T>(key: string, value: T, options: { ttl: number }): Promise<void> {
    await this.redis.setex(key, options.ttl, JSON.stringify(value));
  }

  generateCacheKey(operation: string, content: string, options: any): string {
    // 生成基于内容和选项的哈希键
    const hash = crypto
      .createHash('sha256')
      .update(`${operation}:${content}:${JSON.stringify(options)}`)
      .digest('hex');

    return `ai_cache:${operation}:${hash}`;
  }
}
```

#### 速率限制实现

[Source: docs/architecture.md#API 安全]

**算法**: 滑动窗口速率限制
**目标**: 防止 AI API 滥用和成本失控

```typescript
// packages/ai/src/utils/rateLimit.ts
interface RateLimitRule {
  maxRequests: number;
  windowMs: number;
}

export class RateLimitService {
  private redis: Redis;
  private rules: Map<string, RateLimitRule>;

  constructor(redis: Redis, rules: Record<string, RateLimitRule>) {
    this.redis = redis;
    this.rules = new Map(Object.entries(rules));
  }

  async checkRateLimit(identifier: string, operation: string): Promise<void> {
    const rule = this.rules.get(operation);
    if (!rule) return;

    const key = `rate_limit:${operation}:${identifier}`;
    const now = Date.now();
    const windowStart = now - rule.windowMs;

    // 清理过期记录
    await this.redis.zremrangebyscore(key, 0, windowStart);

    // 获取当前窗口内的请求数
    const currentRequests = await this.redis.zcard(key);

    if (currentRequests >= rule.maxRequests) {
      throw new RateLimitExceededError(`Rate limit exceeded for ${operation}`);
    }

    // 记录当前请求
    await this.redis.zadd(key, now, `${now}-${Math.random()}`);
    await this.redis.expire(key, Math.ceil(rule.windowMs / 1000));
  }
}
```

      tags: {
        maxRequests: parseInt(process.env.AI_TAGS_RATE_LIMIT || '200'),
        windowMs: parseInt(process.env.AI_TAGS_WINDOW_MS || '3600000'),
      },
    },
  };

  // 验证配置
  validateAIConfig(config);

  return config;
}

function validateAIConfig(config: AIConfig): void {

    throw new Error('At least one AI provider must be enabled');

    throw new Error('OpenAI API key is required when OpenAI is enabled');
  }

  if (config.claude?.enabled && !config.claude.apiKey) {
    throw new Error('Claude API key is required when Claude is enabled');
  }
}
