```typescript
// packages/ai/src/providers/OpenAIProvider.ts
export class OpenAIProvider implements AIProvider {
  async generateSummary(
    content: string,
    options: SummaryOptions = {}
  ): Promise<SummaryResult> {
    // 1. 成本检查
    await this.costTracker.checkBudget('summary');

    // 2. 缓存检查
    const cacheKey = this.generateCacheKey('summary', content, options);
    const cached = await this.cache.get(cacheKey);
    if (cached) return cached;

    // 3. API调用
    const response = await this.client.chat.completions.create({
      model: 'gpt-4o-mini',
      messages: [
        {
          role: 'system',
          content: this.getSummarySystemPrompt(options),
        },
        {
          role: 'user',
          content: this.preprocessContent(content, options.maxLength || 4000),
        },
      ],
      max_tokens:
        options.summaryLength === 'short'
          ? 100
          : options.summaryLength === 'long'
            ? 300
            : 200,
      temperature: 0.3,
    });

    // 4. 结果处理和缓存
    const result: SummaryResult = {
      summary: response.choices[0]?.message?.content?.trim(),
      confidence: this.calculateConfidence(response),
      language: await this.detectLanguage(summary),
    };

    await this.cache.set(cacheKey, result, { ttl: 7 * 24 * 3600 });
    return result;
  }
}
```

#### AI 服务统一接口

[Source: docs/architecture.md#AI 服务统一接口]

**故障转移机制**: 主提供商失败时自动切换到备选提供商
**成本控制**: BudgetExceededError 不触发故障转移，其他错误支持降级

```typescript
// packages/ai/src/AIService.ts
export class AIService {
  async generateSummary(
    content: string,
    options: SummaryOptions = {}
  ): Promise<SummaryResult> {
    const provider =
      options.provider || this.config.defaultProvider || 'openai';
    return this.executeWithFallback(
      'generateSummary',
      provider,
      content,
      options
    );
  }

  private async executeWithFallback<T>(
    method: keyof AIProvider,
    primaryProvider: string,
    ...args: any[]
  ): Promise<T> {
    // 尝试主提供商，失败时降级到备选提供商
    for (const providerName of this.getProviderChain(primaryProvider)) {
      try {
        return await (provider[method] as any)(...args);
      } catch (error) {
        if (error instanceof BudgetExceededError) throw error;
        continue; // 其他错误继续尝试下一个提供商
      }
    }
    throw new AIError('All AI providers failed');
  }
}
```

```typescript
// packages/ai/src/prompts/summary.ts
export class SummaryPrompts {
  getSummarySystemPrompt(options: SummaryOptions): string {
    const language = options.language || 'en';
    const length = options.summaryLength || 'medium';

    const basePrompt = this.getLanguageSpecificPrompt(language);
    const lengthGuideline = this.getLengthGuideline(length, language);

    return `${basePrompt}\n\n${lengthGuideline}\n\n请确保摘要：
1. 准确反映原文主要内容和观点
2. 使用清晰简洁的语言
3. 保持客观中性的语调  
4. 避免引入原文中没有的信息
5. 符合指定的长度要求`;
  }

  private getLanguageSpecificPrompt(language: string): string {
    const prompts = {
      zh: '你是一个专业的中文内容摘要生成器。请为以下网页内容生成一个准确、简洁的中文摘要。',
      en: 'You are a professional content summarizer. Please generate an accurate and concise summary of the following web content.',
      ja: 'あなたは専門的なコンテンツ要約生成器です。以下のウェブコンテンツの正確で簡潔な要約を生成してください。',
      // 支持更多语言...
    };
    return prompts[language] || prompts['en'];
  }
}
```

```typescript
// packages/ai/src/services/CostTracker.ts
export class CostTracker {
  async checkBudget(
    operation: 'summary' | 'tags' | 'embedding'
  ): Promise<void> {
    const usage = await this.getCurrentMonthUsage();
    const estimates = {
      summary: 0.01, // 估算每次摘要成本
      tags: 0.005, // 估算每次标签成本
      embedding: 0.002, // 估算每次嵌入成本
    };

    if (usage.totalCost + estimates[operation] > this.monthlyBudget) {
      throw new BudgetExceededError(`Monthly budget exceeded for ${operation}`);
    }
  }

  async recordUsage(operation: string, usage: UsageMetrics): Promise<void> {
    await this.redis.zadd(
      `cost:${this.userId}:${this.getCurrentMonth()}`,
      Date.now(),
      JSON.stringify({ operation, ...usage })
    );
  }
}
```
