```typescript
// packages/ai/src/utils/ContentOptimizer.ts
export class ContentOptimizer {
  private costTracker: CostTracker;

  constructor(costTracker: CostTracker) {
    this.costTracker = costTracker;
  }

  async optimizeForAI(
    content: ReadableContent,
    maxTokens: number = 3000
  ): Promise<OptimizedContent> {
    // 1. 估算当前内容的token成本
    const estimatedTokens = this.estimateTokens(content.textContent);

    if (estimatedTokens <= maxTokens) {
      return {
        content: content.textContent,
        truncated: false,
        originalLength: estimatedTokens,
        optimizedLength: estimatedTokens,
      };
    }

    // 2. 智能截断保持语义完整性
    const optimizedContent = this.intelligentTruncate(content, maxTokens);

    // 3. 记录优化统计
    await this.costTracker.recordOptimization({
      originalTokens: estimatedTokens,
      optimizedTokens: this.estimateTokens(optimizedContent),
      truncationRatio: optimizedContent.length / content.textContent.length,
    });

    return {
      content: optimizedContent,
      truncated: true,
      originalLength: estimatedTokens,
      optimizedLength: this.estimateTokens(optimizedContent),
    };
  }

  private intelligentTruncate(
    content: ReadableContent,
    maxTokens: number
  ): string {
    // 优先保留：
    // 1. 标题和重要段落
    // 2. 列表和结构化信息
    // 3. 文章开头和结尾
    // 4. 在句子边界截断
  }

  private estimateTokens(text: string): number {
    // 简单的token估算：大约4个字符=1个token
    // 可以后续集成更精确的tokenizer
    return Math.ceil(text.length / 4);
  }
```

```typescript
// packages/ai/src/types/extraction.ts
export class ExtractionError extends Error {
  constructor(
    message: string,
    public readonly code: ExtractionErrorCode,
    public readonly originalError?: Error
  ) {
    super(message);
    this.name = 'ExtractionError';
  }
}

export enum ExtractionErrorCode {
  READABILITY_FAILED = 'READABILITY_FAILED',
  HTML_PARSE_FAILED = 'HTML_PARSE_FAILED',
  CONTENT_TOO_SHORT = 'CONTENT_TOO_SHORT',
  NETWORK_ERROR = 'NETWORK_ERROR',
  TIMEOUT = 'TIMEOUT',
}

// packages/ai/src/services/ContentExtractionService.ts
export class ContentExtractionService {
  async extractContentWithFallback(url: string): Promise<ExtractedContent> {
    const strategies = [
      () => this.extractWithReadability(url),
      () => this.extractWithBasicParsing(url),
      () => this.extractMetadataOnly(url),
    ];

    let lastError: Error | null = null;

    for (const strategy of strategies) {
      try {
        const result = await strategy();

        // 验证提取质量
        if (this.validateExtractionQuality(result)) {
          return result;
        }
      } catch (error) {
        lastError = error;
        console.warn(`Content extraction strategy failed:`, error.message);
        continue;
      }
    }

    // 所有策略都失败了
    throw new ExtractionError(
      'All content extraction strategies failed',
      ExtractionErrorCode.HTML_PARSE_FAILED,
      lastError || undefined
    );
  }

  private validateExtractionQuality(content: ExtractedContent): boolean {
    // 验证内容质量：
    // - 文本内容长度 > 最小阈值
    // - 标题存在且有意义
    // - 基础元数据完整性
    return content.textContent.length > 100 && content.title.length > 0;
  }
}
```
