## Story

**作为一个** 用户，

async findByUrlAndUser(url: string, userId: string): Promise<Bookmark | null> {
const bookmark = await this.db
.select()
.from(bookmarks)
.where(
and(
eq(bookmarks.url, url),
eq(bookmarks.userId, userId),
eq(bookmarks.isDeleted, false)
)
)
.limit(1);
return bookmark[0] || null;
}

async list(params: ListBookmarksParams): Promise<PaginatedBookmarks> {
const { page, limit, search, tags, sortBy, sortOrder } = params;

    let query = this.db
      .select()
      .from(bookmarks)
      .where(
        and(
          eq(bookmarks.userId, params.userId),
          eq(bookmarks.isDeleted, false)
        )
      );

    if (search) {
      query = query.where(
        or(
          ilike(bookmarks.title, `%${search}%`),
          ilike(bookmarks.description, `%${search}%`),
          ilike(bookmarks.content, `%${search}%`)
        )
      );
    }

    if (tags && tags.length > 0) {
      query = query.where(
        sql`array_length(${bookmarks.tags} && ${tags}, 1) > 0`
      );
    }

    const total = await query.execute().then(rows => rows.length);

    const offset = (page - 1) * limit;
    const bookmarksList = await query
      .orderBy(
        sortOrder === 'desc'
          ? desc(bookmarks[sortBy])
          : asc(bookmarks[sortBy])
      )
      .limit(limit)
      .offset(offset);

    return {
      bookmarks: bookmarksList,
      total,
      page,
      totalPages: Math.ceil(total / limit),
    };

}

async update(id: string, userId: string, data: UpdateBookmarkData): Promise<Bookmark | null> {
const [bookmark] = await this.db
.update(bookmarks)
.set({ ...data, updatedAt: new Date() })
.where(
and(
eq(bookmarks.id, id),
eq(bookmarks.userId, userId),
eq(bookmarks.isDeleted, false)
)
)
.returning();
return bookmark || null;
}

async softDelete(id: string, userId: string): Promise<boolean> {
const [result] = await this.db
.update(bookmarks)
.set({
isDeleted: true,
updatedAt: new Date()
})
.where(
and(
eq(bookmarks.id, id),
eq(bookmarks.userId, userId),
eq(bookmarks.isDeleted, false)
)
)
.returning();
return !!result;
}
}
